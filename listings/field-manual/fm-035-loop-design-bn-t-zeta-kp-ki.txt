theta_n = (Bn*T) / (zeta + 1/(4*zeta))

Kp = (4*zeta*theta_n)   / (1 + 2*zeta*theta_n + theta_n*theta_n) / Kd
Ki = (4*theta_n*theta_n) / (1 + 2*zeta*theta_n + theta_n*theta_n) / Kd
