required_cycles_per_second =
    symbol_rate * (matched_filter_taps * 2      // FIR: multiply + accumulate per tap
                 + timing_loop_update_cost       // Gardner TED + loop filter
                 + carrier_loop_update_cost      // Costas detector + loop filter

    + fec_decode_cost_per_second                 // Viterbi/RS, computed separately
