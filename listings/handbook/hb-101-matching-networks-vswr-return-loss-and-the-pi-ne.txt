Gamma = (Z_antenna - Z_0) / (Z_antenna + Z_0)     // Z_0 is the system impedance, usually 50 ohms

VSWR = (1 + |Gamma|) / (1 - |Gamma|)               // ranges from 1.0 (perfect match) upward
Return_loss_dB = -20 * log10(|Gamma|)              // higher dB = better match
